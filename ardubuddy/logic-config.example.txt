    pin: 5
    skill: gpio.write
    safe_states: [on, off]

  projector_indicator:
    type: output
    pin: 16
    skill: gpio.write
    safe_states: [on, off]

  room_fan:
    type: output
    pin: 17
    skill: gpio.write
    safe_states: [on, off]

    skill: gpio.write
    safe_states: [on, off]
